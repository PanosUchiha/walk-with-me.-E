cmake_minimum_required(VERSION 3.24)
project(BTPSSceneGraphGuard VERSION 0.1.0 LANGUAGES CXX)
find_package(CommonLibSSE CONFIG REQUIRED)
add_library(BTPSSceneGraphGuard SHARED plugin.cpp)
target_compile_features(BTPSSceneGraphGuard PRIVATE cxx_std_23)
target_compile_definitions(BTPSSceneGraphGuard PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(BTPSSceneGraphGuard PRIVATE /W4 /permissive- /Zc:__cplusplus)
target_link_libraries(BTPSSceneGraphGuard PRIVATE CommonLibSSE::CommonLibSSE bcrypt)
target_link_options(BTPSSceneGraphGuard PRIVATE /DEBUG:FULL)
set_target_properties(BTPSSceneGraphGuard PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
add_executable(GuardTests tests.cpp)
target_compile_features(GuardTests PRIVATE cxx_std_23)
target_compile_definitions(GuardTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(GuardTests PRIVATE /W4 /permissive-)
target_link_libraries(GuardTests PRIVATE bcrypt)
add_test(NAME BTPSSceneGraphGuard COMMAND GuardTests)
